R:   1 | + /**
R:   2 | +  * Settlement receipts — the signed artifact that authorises a local mint (#104, step 3b).
R:   3 | +  *
R:   4 | +  * Spec: docs/federation-economics.md §2.5, Rule 3c ("a node never credits a local member before it holds
R:   5 | +  * a signed settlement receipt from the buyer's node").
R:   6 | +  *
R:   7 | +  * WHY SIGN AT ALL, when the stream is already an authenticated Noise channel from a trusted peer?
R:   8 | +  * Because the receipt has to outlive the connection. The seller's node persists it and may act on it
R:   9 | +  * minutes later, or after a reboot, with no live connection to re-authenticate against. At that point the
R:  10 | +  * signature is the *only* evidence that the buyer's node really committed. Transport authentication
R:  11 | +  * proves who is talking now; the signature proves who committed then, and stays checkable in an audit.
R:  12 | +  *
R:  13 | +  * WHAT THE SIGNATURE BINDS, and why each field is in it:
R:  14 | +  *   • `key`            — so a receipt cannot be moved onto a different settlement
R:  15 | +  *   • `issuerPeerId`   — so a receipt cannot be replayed by a DIFFERENT peer. This one matters most: the
R:  16 | +  *                        bridge account we debit is chosen by who issued the receipt, so an unbound
R:  17 | +  *                        receipt would let peer B spend peer A's credit line.
R:  18 | +  *   • `amount`         — so a receipt cannot be inflated past the reserved (and cap-checked) figure
R:  19 | +  *   • buyer/seller/post — so the seller's node can confirm it is paying for the trade it reserved
R:  20 | +  *
R:  21 | +  * The signing key is the node's persistent libp2p Ed25519 identity (`p2p.ts`), so verification needs no
R:  22 | +  * new key distribution: an Ed25519 PeerId *embeds* its public key, and the peer id is already what the
R:  23 | +  * connector trust list is keyed on. Nothing new to configure, nothing new to get wrong.
R:  24 | +  */
R:  25 | + 
R:  26 | + import { peerIdFromString } from '@libp2p/peer-id';
R:  27 | + 
R:  28 | + /**
R:  29 | +  * Domain separator. The node identity key also signs libp2p handshakes and (in the DNS registrar work)
R:  30 | +  * attestations, so a bare signature over bare fields could in principle be lifted from one context into
R:  31 | +  * another. Prefixing every payload with a context string and version makes that impossible, and gives us
R:  32 | +  * a way to change the format later without ambiguity.
R:  33 | +  */
R:  34 | + const RECEIPT_DOMAIN = 'beanpool-settlement-receipt/1';
R:  35 | + 
R:  36 | + export interface SettlementReceipt {
R:  37 | +     /** The idempotency key, minted by the buyer's node. */
R:  38 | +     key: string;
R:  39 | +     /** libp2p peer id of the node that committed — the buyer's node. Bound into the signature. */
R:  40 | +     issuerPeerId: string;
R:  41 | +     buyerPublicKey: string;
R:  42 | +     /** Public URL of the buyer's home node, for the seller's visitor record. May be absent. */
R:  43 | +     buyerHomeNode: string | null;
R:  44 | +     sellerPublicKey: string;
R:  45 | +     postId: string | null;
R:  46 | +     /** The agreed price — NEVER price plus fee. The fee never crosses the border (§2.1). */
R:  47 | +     amount: number;
R:  48 | +     /** ISO timestamp of the commit on the buyer's node. */
R:  49 | +     committedAt: string;
R:  50 | + }
R:  51 | + 
R:  52 | + /**
R:  53 | +  * Canonicalise a number for signing.
R:  54 | +  *
R:  55 | +  * Both sides rebuild the payload from parsed fields, so the formatting must be identical on both or every
R:  56 | +  * signature fails. Fixed decimal places rather than `String(n)`, because `String()` renders 5 as "5" and
R:  57 | +  * 5.0 as "5" but 1e21 as "1e+21" — a representation that varies with magnitude is exactly what a
R:  58 | +  * canonical form must not have.
R:  59 | +  */
R:  60 | + const canonicalAmount = (n: number): string => n.toFixed(4);
R:  61 | + 
R:  62 | + /**
R:  63 | +  * The exact bytes that get signed.
R:  64 | +  *
R:  65 | +  * Built from an explicit ordered field list rather than `JSON.stringify(receipt)`, because object key
R:  66 | +  * order is a property of how the object was constructed — a receipt rebuilt from a parsed wire message
R:  67 | +  * could serialise its keys in a different order and fail to verify, intermittently and confusingly.
R:  68 | +  */
R:  69 | + export function receiptPayload(r: SettlementReceipt): string {
R:  70 | +     return [

R:  72 | +         r.key,
R:  73 | +         r.issuerPeerId,
R:  74 | +         r.buyerPublicKey,
R:  75 | +         r.buyerHomeNode ?? '',
R:  76 | +         r.sellerPublicKey,
R:  77 | +         r.postId ?? '',
R:  78 | +         canonicalAmount(r.amount),
R:  79 | +         r.committedAt,
R:  80 | +     ].join('\n');
R:  81 | + }
R:  82 | + 
R:  83 | + /** Sign a receipt with this node's libp2p identity key. Returns base64. */
R:  84 | + export async function signReceipt(r: SettlementReceipt, privateKey: any): Promise<string> {
R:  85 | +     if (!privateKey?.sign) throw new Error('No node identity key available to sign a settlement receipt');
R:  86 | +     const sig = await privateKey.sign(new TextEncoder().encode(receiptPayload(r)));
R:  87 | +     return Buffer.from(sig).toString('base64');
R:  88 | + }
R:  89 | + 
R:  90 | + /**
R:  91 | +  * Verify a receipt against the peer that presented it.
R:  92 | +  *
R:  93 | +  * `expectedIssuerPeerId` must be the remote peer of the *connection the receipt arrived on* (or, during
R:  94 | +  * boot recovery, the peer recorded on the settlement row). Passing the receipt's own `issuerPeerId` here
R:  95 | +  * would make the check circular and worthless — the whole point is to pin the claim to an independently
R:  96 | +  * known identity.
R:  97 | +  *
R:  98 | +  * Fails closed on every unexpected shape: a malformed peer id, a peer id type that does not embed a
R:  99 | +  * public key, a bad base64 signature. Never throws, so a caller cannot accidentally treat an exception
R: 100 | +  * path as success.
R: 101 | +  */
R: 102 | + export async function verifyReceipt(
R: 103 | +     r: SettlementReceipt,
R: 104 | +     signatureB64: string,
R: 105 | +     expectedIssuerPeerId: string,
R: 106 | + ): Promise<boolean> {
R: 107 | +     try {
R: 108 | +         if (!r?.key || !r.issuerPeerId || !signatureB64) return false;
R: 109 | +         if (r.issuerPeerId !== expectedIssuerPeerId) return false;
R: 110 | + 
R: 111 | +         const peerId: any = peerIdFromString(r.issuerPeerId);
R: 112 | +         // RSA peer ids carry only a hash of the key, so the key is not recoverable from the id alone.
R: 113 | +         // We only ever generate Ed25519 identities; refuse rather than guess.
R: 114 | +         if (!peerId.publicKey?.verify) return false;
R: 115 | + 
R: 116 | +         const sig = Buffer.from(signatureB64, 'base64');
R: 117 | +         if (sig.length === 0) return false;
R: 118 | + 
R: 119 | +         return await peerId.publicKey.verify(new TextEncoder().encode(receiptPayload(r)), sig);
R: 120 | +     } catch {
R: 121 | +         return false;
R: 122 | +     }
R: 123 | + }
